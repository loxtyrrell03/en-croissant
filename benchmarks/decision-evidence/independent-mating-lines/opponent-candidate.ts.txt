type ReplyGain = {
    witnesses: TacticalSequenceWitness[];
    description: string;
    moveUci: string;
    outcome: string[];
    materialPoints: number;
};

function replyEvidence(fen: string, line: DecisionEngineLine, root: TacticalReplayStep): ReplyGain[] {
    const measured = measureDecisionLine(fen, line);
    if (!measured || measured.drawnEnding) return [];
    const finite = { ...line, pvUci: line.pvUci.slice(0, 12) };
    const next = replayTacticalLine(fen, finite.pvUci);
    const [first] = next;
    if (!first) return [];
    const supported = extractTacticalSequenceWitnesses(fen, finite).filter(
        (item) => item.rootMoveInvolved && item.actor === root.after.turn &&
            item.mechanismPly <= 2 && item.payoffPly <= 5 && item.mechanism !== "tacticalPreparation",
    );
    const results = supported.flatMap((item) => {
        const outcome = measureTacticalEpisode(fen, finite, item);
        if (!outcome || outcome.materialGainPoints * 100 - root.balance < 90) return [];
        return [{
            witnesses: [item],
            description: item.description,
            moveUci: first.uci,
            outcome: [root.uci, ...outcome.lineUci],
            materialPoints: (outcome.materialGainPoints * 100 - root.balance) / 100,
        }];
    });
    if (results.length || !first.capture) return results;
    const localGain = tacticalCaptureGain(first, 2048);
    if (localGain == null || localGain - root.balance < 190) return results;
    const resultWindow = next.findIndex((_, i) =>
        i >= 3 && next.slice(i - 3, i + 1).every((step) => step.balance === next[i].balance) &&
        next.slice(i - 1, i + 1).every((step) => !step.capture && !step.after.isCheck()) &&
        next[i].balance - root.balance >= 190,
    );
    if (resultWindow < 0) return results;
    return [{
        witnesses: [],
        description: first.san + " wins material in the ensuing exchange.",
        moveUci: first.uci,
        outcome: [root.uci, ...finite.pvUci.slice(0, resultWindow + 1)],
        materialPoints: (next[resultWindow].balance - root.balance) / 100,
    }];
}

function normalReply(fen: string, candidate: DecisionEngineLine) {
    if (candidate.depth < 16 || candidate.pvUci.length < 2 || !measureDecisionLine(fen, candidate))
        return null;
    const [root] = replayTacticalLine(fen, candidate.pvUci.slice(0, 1));
    if (!root) return null;
    const child = makeFen(root.after.toSetup());
    return { root, child, reply: { ...candidate, fen: child, cp: candidate.cp === null ? null : -candidate.cp,
        mate: candidate.mate == null ? null : -candidate.mate, pvUci: candidate.pvUci.slice(1) } };
}

function sameConsequence(a: ReplyGain, b: ReplyGain) {
    if (a.moveUci !== b.moveUci || Math.abs(a.materialPoints - b.materialPoints) > 0.5) return false;
    return a.witnesses.length === 0
        ? b.witnesses.length === 0
        : a.witnesses.some((left) => b.witnesses.some((right) => left.mechanism === right.mechanism &&
            left.squares.length === right.squares.length &&
            left.squares.every((square) => right.squares.includes(square))));
}

export function compareOpponentTactics(
    fen: string,
    candidates: DecisionEngineLine[],
    replies: DecisionEngineLine[],
) {
    const [best] = candidates;
    const empty = { confirmedReply: null, repeatedRoot: null, forcedMate: null };
    if (!best) return empty;
    const leading = normalReply(fen, best);
    const [independent] = replies;
    if (!leading || !independent || independent.depth < 16 ||
        !measurePositionDecision(leading.child, replies) ||
        independent.pvUci[0] !== leading.reply.pvUci[0]) return empty;
    const { root, child, reply } = leading;
    let forcedMate: PositionStrategicPlan | null = null;
    if (reply.mate != null && reply.mate > 0 && reply.mate <= 6 &&
        independent.mate != null && independent.mate > 0 && independent.mate <= 6 &&
        measureDecisionLine(child, independent)?.checkmate) {
        forcedMate = {
            linesUci: [best.pvUci, [root.uci, ...independent.pvUci]],
            description: "After " + root.san + ", both searches find a short mate for the opponent starting with " +
                replayTacticalLine(child, independent.pvUci.slice(0, 1))[0].san + ".",
        };
    }
    if (reply.cp === null || independent.cp === null || Math.abs(reply.cp - independent.cp) > 90)
        return { ...empty, forcedMate };
    const main = replyEvidence(child, reply, root);
    if (!main.length) return empty;
    const independentOptions = replyEvidence(child, independent, root);
    for (const evidence of main) {
        const corroboration = independentOptions.find((item) => sameConsequence(evidence, item));
        if (!corroboration) continue;
        const confirmedReply: PositionStrategicPlan = {
            linesUci: [evidence.outcome, corroboration.outcome],
            description: "After " + root.san + ", " + evidence.description,
        };
        for (const alternative of candidates.slice(1, 3)) {
            if (alternative.cp === null || alternative.depth !== best.depth ||
                best.cp! - alternative.cp < 0 || best.cp! - alternative.cp > 65) continue;
            const branch = normalReply(fen, alternative);
            if (!branch || branch.reply.pvUci[0] !== evidence.moveUci) continue;
            const repetition = replyEvidence(branch.child, branch.reply, branch.root).find(
                (item) => sameConsequence(evidence, item),
            );
            if (!repetition) continue;
            const repeatedRoot: PositionStrategicPlan = {
                linesUci: [corroboration.outcome, repetition.outcome],
                description: "After " + root.san + ", " + evidence.description +
                    " The same tactical reply also works after " + branch.root.san + ".",
            };
            return { confirmedReply, repeatedRoot, forcedMate };
        }
        return { ...empty, confirmedReply, forcedMate };
    }
    return empty;
}
type RookFile = {
    key: string;
    file: string;
    pawn: Square | undefined;
    rooks: [Square, Square];
};

function contestedFiles(position: Chess): RookFile[] {
    const result: RookFile[] = [];
    const { board } = position;
    for (const white of board.rook.intersect(board.white)) {
        for (const black of board.rook.intersect(board.black)) {
            if (white % 8 !== black % 8) continue;
            const blocking = between(white, black).intersect(board.occupied);
            if (blocking.size() > 1) continue;
            const pawn = blocking.first();
            if (pawn !== undefined && !board.pawn.has(pawn)) continue;
            const file = makeSquare(white)[0];
            result.push({
                key: file + ":" + (pawn !== undefined ? makeSquare(pawn) : "open"),
                file, pawn, rooks: [white, black],
            });
        }
    }
    return result;
}

type Contest = { start: number; until: number; observation: RookFile; steps: TacticalReplayStep[] };

function fileContests(fen: string, uci: string[]): Contest[] {
    const steps = replayTacticalLine(fen, uci.slice(0, 12));
    if (steps.length < 7) return [];
    const initial = new Set(contestedFiles(steps[0].before).map(({ key }) => key));
    const contests = [];
    for (const [i, step] of steps.slice(0, 6).entries()) {
        // A rook choosing to contest the file after an exchange is distinct
        // from two recaptures incidentally ending on the same file.
        if (step.capture || step.after.isCheck() || step.before.isCheck() ||
            step.before.board.get(step.move.from)?.role !== "rook") continue;
        for (const observation of contestedFiles(step.after)) {
            if (initial.has(observation.key) || !observation.rooks.includes(step.move.to)) continue;
            let until = i + 1;
            for (const later of steps.slice(i + 1)) {
                if (later.balance !== step.balance || later.after.isCheck() ||
                    !contestedFiles(later.after).some(({key}) => key === observation.key)) break;
                until++;
            }
            if (until - i >= 3) contests.push({ start: i + 1, until, observation, steps });
        }
    }
    return contests;
}

function compareRookFileContests(fen: string, candidates: DecisionEngineLine[], replies: DecisionEngineLine[]) {
    const [best] = candidates;
    if (!best || best.cp === null || best.depth < 16) return null;
    const leading = normalReply(fen, best);
    const [independent] = replies;
    if (!leading || !independent || independent.depth < 16 || independent.cp === null ||
        leading.reply.cp === null || Math.abs(leading.reply.cp - independent.cp) > 90 ||
        !measureDecisionLine(leading.child, independent)) return null;
    const rootContests = fileContests(fen, best.pvUci);
    if (!rootContests.length) return null;
    const verified = fileContests(fen, [leading.root.uci, ...independent.pvUci]);
    for (const alternative of candidates.slice(1, 3)) {
        if (alternative.cp === null || alternative.mate != null || alternative.depth !== best.depth ||
            best.cp - alternative.cp > 50 || best.cp < alternative.cp ||
            alternative.pvUci[0] === best.pvUci[0] || !measureDecisionLine(fen, alternative)) continue;
        const other = fileContests(fen, alternative.pvUci);
        for (const main of rootContests) {
            const full = verified.find((x) => x.observation.key === main.observation.key);
            const branch = other.find((x) => x.observation.key === main.observation.key);
            if (!full || !branch) continue;
            const end = Math.max(main.start, full.start, branch.start) + 2;
            if ([main, full, branch].some((x) => x.until < end)) continue;
            const before = new Set(main.steps.slice(0, main.start - 1).map(({after}) => tacticalRepetitionKey(after)));
            if (branch.steps.slice(0, branch.start - 1).some(({after}) => before.has(tacticalRepetitionKey(after))))
                continue;
            if (Math.abs(main.steps[end - 1].balance - full.steps[end - 1].balance) > 50 ||
                Math.abs(main.steps[end - 1].balance - branch.steps[end - 1].balance) > 50) continue;
            const { file, pawn } = main.observation;
            const action = pawn === undefined
                ? "The rooks face each other with an open line between them."
                : "The rooks line up behind and against the pawn on " + makeSquare(pawn) + ".";
            return {
                linesUci: [
                    full.steps.slice(0, end).map((step) => step.uci),
                    branch.steps.slice(0, end).map((step) => step.uci),
                ],
                description: "After " + full.steps[0].san + " or " + branch.steps[0].san +
                    ", opposing rooks contest the " + file + "-file. " + action,
            } satisfies PositionStrategicPlan;
        }
    }
    return null;
}

export { compareRookFileContests };
